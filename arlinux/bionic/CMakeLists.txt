cmake_minimum_required(VERSION 3.16)
project(ARLinuxBionicCapture LANGUAGES C CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Use the Android NDK arm64-v8a toolchain")
endif()
get_filename_component(GFXRECON_SOURCE_DIR "../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 20)
set(GFXRECON_ENABLE_VULKAN ON)
set(OPENXR_SUPPORT_ENABLED FALSE)
set(nlohmann_json_DIR "${GFXRECON_SOURCE_DIR}/external/nlohmann-json/share/cmake/nlohmann_json")
set(WAYLAND_FOUND TRUE)
add_subdirectory(../../android/layer android-layer)
target_compile_definitions(platform_specific INTERFACE VK_USE_PLATFORM_WAYLAND_KHR GFXRECON_BIONIC_ENVIRONMENT)
target_include_directories(platform_specific INTERFACE "${BIONIC_PREFIX}/include")
add_subdirectory(../../framework/plugin plugin)
add_subdirectory(../../framework/decode decode)
add_subdirectory(../../tools/convert convert)
add_subdirectory(../../tools/info info)
